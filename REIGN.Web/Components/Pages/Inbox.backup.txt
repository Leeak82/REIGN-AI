@page "/inbox"

@using System.Net.Http.Json

@inject HttpClient Http

<h3>REIGN Inbox</h3>

@if (!string.IsNullOrEmpty(Error))
{
    <div style="color:red">
        @Error
    </div>
}

<div style="display:flex; gap:20px">

    <div style="width:280px">

        <h4>Customers</h4>

        @foreach (var customer in Customers)
        {
            <button style="width:100%; margin-bottom:8px"
        @onclick="async () => await LoadConversation(customer.PhoneNumber)">

    @customer.PhoneNumber

    <br />

    <small>
        Messages: @customer.Messages |
        Appointments: @customer.Appointments
    </small>

</button>
        }

    </div>


    <div style="flex:1">

        <h4>Conversation</h4>


        @foreach (var message in Messages)
        {
            <div style="
                margin:10px;
                padding:10px;
                border-radius:8px;
                background:@(message.Direction == "Inbound" ? "#eeeeee" : "#bde0ff")">

                <b>@message.Direction</b>

                <br />

                @message.Body

                <br />

                <small>
                    @message.CreatedAt
                </small>

            </div>
        }


        @if (!string.IsNullOrEmpty(SelectedPhone))
        {

            <hr />

            <h4>Reply</h4>


            <textarea 
                style="width:100%;height:80px"
                @bind="ReplyText"
                placeholder="Type reply...">
            </textarea>


            <button 
                style="margin-top:10px"
                @onclick="SendReply">

                Send SMS

            </button>

        }

    </div>

</div>


@code {

    List<CustomerDto> Customers = new();

    List<MessageDto> Messages = new();

    string SelectedPhone = "";

    string ReplyText = "";

    string Error = "";


    string Api = "http://localhost:5204";


    protected override async Task OnInitializedAsync()
    {
        try
        {
            Customers =
                await Http.GetFromJsonAsync<List<CustomerDto>>
                ($"{Api}/api/customers")
                ?? new();
        }
        catch(Exception ex)
        {
            Error = ex.Message;
        }
    }



    async Task LoadConversation(string phone)
{
    try
    {
        SelectedPhone = phone;

        Messages =
            await Http.GetFromJsonAsync<List<MessageDto>>
            ($"{Api}/api/messages/{phone}")
            ?? new();

        StateHasChanged();
    }
    catch(Exception ex)
    {
        Error = ex.Message;
        StateHasChanged();
    }
}



    async Task SendReply()
    {

        if(string.IsNullOrWhiteSpace(ReplyText))
            return;


        try
        {

            await Http.PostAsJsonAsync(
                $"{Api}/api/messages/send",
                new
                {
                    PhoneNumber = SelectedPhone,
                    Body = ReplyText
                });


            ReplyText = "";


            await LoadConversation(SelectedPhone);

        }
        catch(Exception ex)
        {
            Error = ex.Message;
        }

    }



    public class CustomerDto
    {
        public Guid Id {get;set;}

        public string PhoneNumber {get;set;}="";

        public string? Name {get;set;}

        public int Messages {get;set;}

        public int Appointments {get;set;}
    }



    public class MessageDto
    {
        public Guid Id {get;set;}

        public string Customer {get;set;}="";

        public string Direction {get;set;}="";

        public string Body {get;set;}="";

        public DateTime CreatedAt {get;set;}
    }

}
